# Quirky zsh shell-ready wrapper
__quirky_usable_zdotdir() {
  [[ -n "${1:-}" ]] || return 1
  # Quirky's own dir, by marker file or by the shape older builds wrote.
  [[ "$1" != */shell-ready/zsh ]] || return 1
  [[ ! -f "$1/.quirky-shell-wrapper" ]] || return 1
  # A directory holding no zsh startup file at all is not a config root,
  # whoever wrote it — and a stale value pointing at one would stop zsh from
  # ever reading the user's real .zshenv.
  local _quirky_startup_file
  for _quirky_startup_file in .zshenv .zshrc .zprofile .zlogin; do
    [[ -r "$1/$_quirky_startup_file" ]] && return 0
  done
  return 1
}
if __quirky_usable_zdotdir "${QUIRKY_ORIG_ZDOTDIR:-}"; then
  builtin export ZDOTDIR="$QUIRKY_ORIG_ZDOTDIR"
else
  builtin unset ZDOTDIR
fi
builtin unset QUIRKY_ORIG_ZDOTDIR QUIRKY_ZSHENV_SOURCE_DIR
builtin unfunction __quirky_usable_zdotdir
builtin typeset -ga _quirky_shell_features
_quirky_shell_features=(${(s:,:)${QUIRKY_SHELL_FEATURES:-}})
builtin unset QUIRKY_SHELL_FEATURES
# Why QUIRKY_HISTFILE is consumed HERE and not in the deferred hook: a user config
# that replaces precmd_functions wholesale drops the hook, and an exported value
# nothing will ever consume is then inherited by every child of this pane,
# including a nested Quirky (#11146). Captured non-exported, it cannot escape.
builtin typeset -g _quirky_histfile="${QUIRKY_HISTFILE:-}"
builtin unset QUIRKY_HISTFILE
__quirky_has_feature() { (( ${_quirky_shell_features[(Ie)$1]} )) }
__quirky_has_feature identity && printf "\033]777;quirky-shell-start:%s\007" "$$"
__quirky_osc133_precmd() {
  local exit_code=$?
  if [[ -n "${__quirky_in_command:-}" ]]; then
    builtin printf "\033]133;D;%s\007" "$exit_code"
    builtin unset __quirky_in_command
  fi
  builtin printf "\033]133;A\007"
}
__quirky_osc133_preexec() {
  builtin printf "\033]133;C\007"
  # Why typeset -g: a plain assignment here creates a global inside a function,
  # which prints a warning above every command under warn_create_global.
  builtin typeset -g __quirky_in_command=1
}
__quirky_deferred_line_init() {
  builtin emulate -L zsh
  (( ${+functions[__quirky_deferred_init]} )) || return 0
  local __quirky_direct_line_init=0
  [[ "${widgets[zle-line-init]:-}" == user:__quirky_deferred_line_init ]] && __quirky_direct_line_init=1
  __quirky_deferred_init
  if (( __quirky_direct_line_init && ${+widgets[zle-line-init]} )); then
    zle zle-line-init "$@"
  elif [[ "${widgets[zle-line-init]:-}" == user:__quirky_prompt_mark ]]; then
    local __quirky_prev_line_init_fn=""
    __quirky_prompt_mark "$@"
  fi
}
# Why: scheduled callbacks run after user prompt hooks without copying their function metadata.
__quirky_deferred_sched_init() {
  local __quirky_prompt_status=$?
  builtin emulate -L zsh
  (( ${+functions[__quirky_deferred_init]} )) && __quirky_deferred_init
  builtin unset __quirky_deferred_sched_armed
  builtin unfunction __quirky_deferred_sched_init
  return $__quirky_prompt_status
}
__quirky_arm_deferred_line_init() {
  builtin emulate -L zsh
  if [[ "${widgets[zle-line-init]:-}" != user:__quirky_deferred_line_init ]]; then
    if (( ${+widgets[zle-line-init]} )); then
      zle -A zle-line-init __quirky_saved_line_init
    fi
    zle -N zle-line-init __quirky_deferred_line_init
  fi
  if (( ! $+__quirky_deferred_sched_armed )) && builtin zmodload -F zsh/sched b:sched 2>/dev/null; then
    builtin sched +0 __quirky_deferred_sched_init && builtin typeset -g __quirky_deferred_sched_armed=1
  fi
}
__quirky_deferred_init() {
  # Why first: this body runs after the user's own config, so it would otherwise
  # inherit whatever options that config left set. Under NO_UNSET an unset
  # precmd_functions is fatal, and KSH_ARRAYS makes the 1-based feature lookup
  # drop whichever feature is listed first.
  builtin emulate -L zsh
  (( $+_quirky_deferred_init_done )) && return 0
  builtin typeset -g _quirky_deferred_init_done=1
  builtin typeset -g precmd_functions
  if (( ${+widgets[__quirky_saved_line_init]} )); then
    if [[ "${widgets[zle-line-init]:-}" == user:__quirky_deferred_line_init ]]; then
      zle -A __quirky_saved_line_init zle-line-init
    fi
    zle -D __quirky_saved_line_init
  elif [[ "${widgets[zle-line-init]:-}" == user:__quirky_deferred_line_init ]]; then
    zle -D zle-line-init
  fi
  if __quirky_has_feature markers; then
    precmd_functions=(${precmd_functions:/__quirky_deferred_init/__quirky_osc133_precmd})
    (( ${precmd_functions[(Ie)__quirky_osc133_precmd]} )) || precmd_functions+=(__quirky_osc133_precmd)
    preexec_functions=(__quirky_osc133_preexec ${preexec_functions[@]})
  else
    precmd_functions=(${precmd_functions:#__quirky_deferred_init})
  fi
  if [ -n "${QUIRKY_CLI_BIN_DIR:-}" ]; then
    case "${PATH:-}" in
      "$QUIRKY_CLI_BIN_DIR"|"$QUIRKY_CLI_BIN_DIR":*) ;;
      *) export PATH="$QUIRKY_CLI_BIN_DIR${PATH:+:$PATH}" ;;
    esac
  fi
  if __quirky_has_feature overlay; then
    # Why: ~/.zshrc can export the user's default OpenCode config after spawn.
    __quirky_restore_agent_teams_path() {
      [[ -n "${QUIRKY_AGENT_TEAMS_SHIM_DIR:-}" ]] || return 0
      case "$PATH" in
        "${QUIRKY_AGENT_TEAMS_SHIM_DIR}"|"${QUIRKY_AGENT_TEAMS_SHIM_DIR}:"*) return 0 ;;
      esac
      export PATH="${QUIRKY_AGENT_TEAMS_SHIM_DIR}:$PATH"
    }
    __quirky_restore_agent_teams_path
    [[ -n "${QUIRKY_OPENCODE_CONFIG_DIR:-}" ]] && export OPENCODE_CONFIG_DIR="${QUIRKY_OPENCODE_CONFIG_DIR}"
    if [[ -n "${QUIRKY_DATA_ACCOUNT_DATA_HOME:-}" ]]; then
      export XDG_DATA_HOME="${QUIRKY_DATA_ACCOUNT_DATA_HOME}"
      export XDG_STATE_HOME="${QUIRKY_DATA_ACCOUNT_STATE_HOME}"
      if [[ "${QUIRKY_DATA_ACCOUNT_PROVIDER:-}" == opencode ]]; then
        export OPENCODE_AUTH_CONTENT=""
        export OPENCODE_DB="opencode.db"
      fi
    fi
    [[ -n "${QUIRKY_MIMOCODE_HOME:-}" ]] && export MIMOCODE_HOME="${QUIRKY_MIMOCODE_HOME}"
    # Why: OMP does not auto-load Quirky's managed status extension; wrap only
    # interactive launch invocations so subcommands such as `omp config` keep
    # their normal argv shape.
    __quirky_omp_should_skip_extension() {
      case "${1:-}" in
        'help'|'--help'|'-h'|'--version'|'-v') return 0 ;;
        '__complete'|'acp'|'agents'|'auth-broker'|'auth-gateway'|'bench'|'commit'|'completions'|'config'|'dry-balance'|'gallery'|'grep'|'grievances'|'install'|'join'|'models'|'plugin'|'read'|'say'|'search'|'setup'|'shell'|'ssh'|'stats'|'tiny-models'|'token'|'ttsr'|'update'|'usage'|'worktree'|'q'|'wt') return 0 ;;
      esac
      return 1
    }
    __quirky_omp_cwd_is_usable() {
      local __quirky_physical_cwd
      [[ -x . ]] || return 1
      if [[ -n "${PWD:-}" && -d "${PWD:-}" ]]; then
        [[ "${PWD}" -ef . ]]
      else
        # Why compare the path: shell builtins can print a cached path for a deleted cwd.
        __quirky_physical_cwd="$(builtin pwd -P 2>/dev/null)" || return 1
        [[ -d "$__quirky_physical_cwd" && "$__quirky_physical_cwd" -ef . ]]
      fi
    }
    __quirky_omp_invoke() {
      local __quirky_use_extension="$1"
      shift
      if [[ $__quirky_use_extension -eq 1 && -n "${QUIRKY_OMP_STATUS_EXTENSION:-}" && -f "${QUIRKY_OMP_STATUS_EXTENSION}" ]]; then
        if [[ "${1:-}" == "launch" ]]; then
          shift
          command omp launch --extension "${QUIRKY_OMP_STATUS_EXTENSION}" "$@"
        else
          command omp --extension "${QUIRKY_OMP_STATUS_EXTENSION}" "$@"
        fi
      else
        command omp "$@"
      fi
    }
    __quirky_omp() {
      local __quirky_use_extension=1
      __quirky_omp_should_skip_extension "${1:-}" && __quirky_use_extension=0
      if ! __quirky_omp_cwd_is_usable; then
        local __quirky_logical_cwd="${PWD:-${QUIRKY_WORKTREE_PATH:-${QUIRKY_ROOT_PATH:-}}}"
        # Why: a restored shell can retain the deleted directory inode after its path is recreated.
        (
          if [[ -z "$__quirky_logical_cwd" ]]; then
            printf 'Quirky: OMP cannot start because no terminal working directory is available. Open a new terminal in an existing directory.\n' >&2
            return 1
          fi
          if ! builtin cd -P -- "$__quirky_logical_cwd" 2>/dev/null; then
            printf 'Quirky: OMP cannot access the terminal working directory "%s". Open a new terminal in an existing directory.\n' "$__quirky_logical_cwd" >&2
            return 1
          fi
          __quirky_omp_invoke "$__quirky_use_extension" "$@"
        )
      else
        __quirky_omp_invoke "$__quirky_use_extension" "$@"
      fi
    }
    if [[ -n "${QUIRKY_OMP_STATUS_EXTENSION:-}" ]]; then
      # Why the function reserved word: it suppresses alias expansion of the name, which
      # an `alias omp` otherwise rewrites at parse time, aborting the rest of the file.
      function omp { __quirky_omp "$@"; }
    fi

    # Why: Codex must keep using Quirky's runtime CODEX_HOME after rc files.
    [[ -n "${QUIRKY_CODEX_HOME:-}" ]] && export CODEX_HOME="${QUIRKY_CODEX_HOME}"
  fi
  # Why: a typed alias expands inside the shell, after pane launch prep.
  # Why unalias inside the substitution: an alias named codex makes command -v
  # report the alias text, and the subshell leaves the user's own alias intact.
  # Why || : twice — zsh alone aborts inside the substitution, but every shell's
  # assignment adopts its exit status, so an absent codex trips set -e in bash too.
  __quirky_codex_binary="$(unalias codex 2>/dev/null || :; command -v codex 2>/dev/null || :)"
  if [[ -n "${__quirky_codex_binary:-}" && -x "${__quirky_codex_binary}" ]]; then
    # Why the function reserved word: it suppresses alias expansion of the name,
    # which otherwise rewrites this header at parse time and aborts the whole file.
    function codex {
      # Why local: zsh's warn_create_global warns for each global a function creates.
      local __quirky_codex_arg __quirky_codex_isolate="${QUIRKY_CODEX_ISOLATE:-1}" __quirky_codex_identity="${QUIRKY_CODEX_ISOLATE:-1}"
      if [[ -n "${QUIRKY_CODEX_LAUNCH_PREFLIGHT:-}" && -x "${QUIRKY_CODEX_LAUNCH_PREFLIGHT}" ]]; then
        "${QUIRKY_CODEX_LAUNCH_PREFLIGHT}" agent hooks prepare-codex >/dev/null 2>&1 || :
      fi
      for __quirky_codex_arg in "$@"; do
        case "$__quirky_codex_arg" in
          --no-daemon) __quirky_codex_isolate=0 ;;
          agents|queue|--remote|--remote=*) __quirky_codex_isolate=0 __quirky_codex_identity=0 ;;
        esac
      done
      if [[ "$__quirky_codex_identity" != 0 ]]; then
        [[ -z "${QUIRKY_TERMINAL_HANDLE:-}" || "${QUIRKY_TERMINAL_HANDLE}" == *"'"* ]] || set -- -c "shell_environment_policy.set.QUIRKY_TERMINAL_HANDLE='${QUIRKY_TERMINAL_HANDLE}'" "$@"
        [[ -z "${QUIRKY_PANE_KEY:-}" || "${QUIRKY_PANE_KEY}" == *"'"* ]] || set -- -c "shell_environment_policy.set.QUIRKY_PANE_KEY='${QUIRKY_PANE_KEY}'" "$@"
        [[ -z "${QUIRKY_WORKTREE_ID:-}" || "${QUIRKY_WORKTREE_ID}" == *"'"* ]] || set -- -c "shell_environment_policy.set.QUIRKY_WORKTREE_ID='${QUIRKY_WORKTREE_ID}'" "$@"
        [[ -z "${QUIRKY_WORKSPACE_ID:-}" || "${QUIRKY_WORKSPACE_ID}" == *"'"* ]] || set -- -c "shell_environment_policy.set.QUIRKY_WORKSPACE_ID='${QUIRKY_WORKSPACE_ID}'" "$@"
        [[ -z "${QUIRKY_USER_DATA_PATH:-}" || "${QUIRKY_USER_DATA_PATH}" == *"'"* ]] || set -- -c "shell_environment_policy.set.QUIRKY_USER_DATA_PATH='${QUIRKY_USER_DATA_PATH}'" "$@"
      fi
      # Why probe every launch: a cached answer goes stale across an upgrade, and 0.155 and older exit 2 on the flag.
      if [[ "$__quirky_codex_isolate" != 0 ]]; then
        case "$(command codex --help 2>/dev/null </dev/null)" in *--no-daemon*) set -- --no-daemon "$@" ;; esac
      fi
      command codex "$@"
    }
  fi
  unset __quirky_codex_binary
  __quirky_claude_binary="$(unalias claude 2>/dev/null || :; command -v claude 2>/dev/null || :)"
  if [[ -n "${QUIRKY_CLAUDE_PROFILE_POINTER:-}" && -n "${__quirky_claude_binary:-}" && -x "${__quirky_claude_binary}" ]]; then
    function claude {
      local __quirky_claude_home __quirky_claude_pointer="${QUIRKY_CLAUDE_PROFILE_POINTER:-}"
      # Why: a WSL pane's pointer is relative to the guest home, which the host cannot know at spawn.
      case "$__quirky_claude_pointer" in '~/'*) __quirky_claude_pointer="${HOME:-}/${__quirky_claude_pointer#??}" ;; esac
      __quirky_claude_home="$(cat "$__quirky_claude_pointer" 2>/dev/null || :)"
      if [ -n "${CLAUDE_CONFIG_DIR:-}" ] && [ "$CLAUDE_CONFIG_DIR" != "${QUIRKY_CLAUDE_INJECTED_CONFIG_DIR:-}" ] &&
        { [ -z "${QUIRKY_CLAUDE_USER_CONFIG_DIR:-}" ] || [ "${CLAUDE_CONFIG_DIR%/}" != "${QUIRKY_CLAUDE_USER_CONFIG_DIR%/}" ]; }; then
        command claude "$@"; return
      fi
      # Why: Quirky's value replaced the user's own at spawn, so System default restores theirs.
      if [ -z "$__quirky_claude_home" ] && [ -n "${CLAUDE_CONFIG_DIR:-}" ] && [ -n "${QUIRKY_CLAUDE_USER_CONFIG_DIR:-}" ]; then
        ( unset QUIRKY_CLAUDE_INJECTED_CONFIG_DIR; export CLAUDE_CONFIG_DIR="$QUIRKY_CLAUDE_USER_CONFIG_DIR"; command claude "$@" ); return
      fi
      if [ -z "$__quirky_claude_home" ]; then
        ( unset CLAUDE_CONFIG_DIR QUIRKY_CLAUDE_INJECTED_CONFIG_DIR; command claude "$@" ); return
      fi
      case "$__quirky_claude_home" in /*|[A-Za-z]:*) ;; *) return 1 ;; esac
      # Why -m 700: matches the folder Quirky's setup creates; a credentials folder stays private.
      [ -d "$__quirky_claude_home" ] || mkdir -p -m 700 -- "$__quirky_claude_home" 2>/dev/null
      ( export CLAUDE_CONFIG_DIR="$__quirky_claude_home" QUIRKY_CLAUDE_INJECTED_CONFIG_DIR="$__quirky_claude_home"; command claude "$@" )
    }
  fi
  unset __quirky_claude_binary
  if [[ -n "${_quirky_histfile:-}" ]]; then
    HISTFILE="$_quirky_histfile"
  fi
  if __quirky_has_feature ready || __quirky_has_feature startup; then
    __quirky_emit_ready_marker=""
    __quirky_has_feature ready && __quirky_emit_ready_marker=1
    # Why: capture the prior zle-line-init so the marker chains to it. On a
    # re-source we are already the bound widget, so keep the function captured
    # the first time instead of clobbering it to empty (which would silently
    # drop the user's widget on every prompt after the second source). Only
    # user-defined widgets are chainable as plain functions; builtin/completion
    # forms (rare for zle-line-init) are left unchained.
    if [[ "${widgets[zle-line-init]:-}" == "user:__quirky_prompt_mark" ]]; then
      :
    elif (( ${+widgets[zle-line-init]} )) && [[ "${widgets[zle-line-init]}" == user:* ]]; then
      __quirky_prev_line_init_fn="${widgets[zle-line-init]#user:}"
    else
      __quirky_prev_line_init_fn=""
    fi
    __quirky_prompt_mark() {
      if [[ "${__quirky_emit_ready_marker-1}" == 1 ]]; then
        printf "\033]777;quirky-shell-ready\007"
      fi
      # Why: call the prior hook as a plain function, not an aliased widget, so
      # $WIDGET stays zle-line-init for add-zle-hook-widget dispatchers.
      if [[ -n "${__quirky_prev_line_init_fn:-}" ]]; then
        "${__quirky_prev_line_init_fn}" "$@"
      fi
      if (( ${+QUIRKY_POSIX_SHELL_STARTUP_COMMAND} )); then
        BUFFER="$QUIRKY_POSIX_SHELL_STARTUP_COMMAND"
        CURSOR=${#BUFFER}
        builtin unset QUIRKY_POSIX_SHELL_STARTUP_COMMAND
        zle accept-line
      fi
    }
    zle -N zle-line-init __quirky_prompt_mark
  fi
  # Why called here: we were appended during this prompt's own precmd sweep, so
  # the permanent hook has not run yet and the first prompt would lose its mark.
  __quirky_has_feature markers && __quirky_osc133_precmd
  builtin unset _quirky_shell_features _quirky_histfile
  (( $+__quirky_deferred_sched_armed )) || builtin unfunction __quirky_deferred_sched_init
  builtin unfunction __quirky_deferred_init __quirky_has_feature __quirky_arm_deferred_line_init
}
{
  builtin typeset _quirky_user_zshenv="${ZDOTDIR-$HOME}/.zshenv"
  [[ ! -r "$_quirky_user_zshenv" ]] || builtin source -- "$_quirky_user_zshenv"
} always {
  builtin unset _quirky_user_zshenv
  if (( ! $+_quirky_deferred_init_done )); then
    builtin typeset -ag precmd_functions
    (( ${precmd_functions[(Ie)__quirky_deferred_init]} )) || precmd_functions+=(__quirky_deferred_init)
    __quirky_arm_deferred_line_init
  fi
}
