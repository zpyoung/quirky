function __quirky_shell_ready_marker --on-event fish_prompt
  builtin printf "\033]777;quirky-shell-ready\007"
  functions -e __quirky_shell_ready_marker
end
# Why captured: an unquoted (type -t codex) expands to zero words when codex is
# absent, leaving "test = file" — fish then errors instead of failing closed.
# Quoting in place is not the fix; fish never substitutes inside double quotes.
set -l __quirky_codex_type (type -t codex 2>/dev/null)
if test "$__quirky_codex_type" = file
  function codex
    if test -x "$QUIRKY_CODEX_LAUNCH_PREFLIGHT"
      command "$QUIRKY_CODEX_LAUNCH_PREFLIGHT" agent hooks prepare-codex >/dev/null 2>&1; or true
    end
    if test "$QUIRKY_CODEX_ISOLATE" != 0; and not string match -qr -- '^(agents|queue|--remote|--remote=.*)$' $argv
      test -n "$QUIRKY_TERMINAL_HANDLE"; and not string match -q -- "*'*" "$QUIRKY_TERMINAL_HANDLE"; and set argv -c "shell_environment_policy.set.QUIRKY_TERMINAL_HANDLE='$QUIRKY_TERMINAL_HANDLE'" $argv
      test -n "$QUIRKY_PANE_KEY"; and not string match -q -- "*'*" "$QUIRKY_PANE_KEY"; and set argv -c "shell_environment_policy.set.QUIRKY_PANE_KEY='$QUIRKY_PANE_KEY'" $argv
      test -n "$QUIRKY_WORKTREE_ID"; and not string match -q -- "*'*" "$QUIRKY_WORKTREE_ID"; and set argv -c "shell_environment_policy.set.QUIRKY_WORKTREE_ID='$QUIRKY_WORKTREE_ID'" $argv
      test -n "$QUIRKY_WORKSPACE_ID"; and not string match -q -- "*'*" "$QUIRKY_WORKSPACE_ID"; and set argv -c "shell_environment_policy.set.QUIRKY_WORKSPACE_ID='$QUIRKY_WORKSPACE_ID'" $argv
      test -n "$QUIRKY_USER_DATA_PATH"; and not string match -q -- "*'*" "$QUIRKY_USER_DATA_PATH"; and set argv -c "shell_environment_policy.set.QUIRKY_USER_DATA_PATH='$QUIRKY_USER_DATA_PATH'" $argv
    end
    if test "$QUIRKY_CODEX_ISOLATE" != 0; and not string match -qr -- '^(agents|queue|--remote|--no-daemon|--remote=.*)$' $argv; and command codex --help 2>/dev/null </dev/null | string match -q -- '*--no-daemon*'
      set argv --no-daemon $argv
    end
    command codex $argv
  end
end
set -e __quirky_codex_type
set -l __quirky_claude_type (type -t claude 2>/dev/null)
if test -n "$QUIRKY_CLAUDE_PROFILE_POINTER"; and test "$__quirky_claude_type" = file
  function claude
    # Why: a WSL pane's pointer is relative to the guest home, which the host cannot know at spawn.
    set -l pointer (string replace -r '^~/' "$HOME/" -- "$QUIRKY_CLAUDE_PROFILE_POINTER")
    set -l profile (cat "$pointer" 2>/dev/null)
    set -l own (string trim -r -c / -- "$QUIRKY_CLAUDE_USER_CONFIG_DIR")
    set -l current (string trim -r -c / -- "$CLAUDE_CONFIG_DIR")
    if test -n "$CLAUDE_CONFIG_DIR"; and test "$CLAUDE_CONFIG_DIR" != "$QUIRKY_CLAUDE_INJECTED_CONFIG_DIR"; and begin; test -z "$own"; or test "$current" != "$own"; end
      command claude $argv
      return $status
    end
    if test -z "$profile"; and test -n "$CLAUDE_CONFIG_DIR"; and test -n "$QUIRKY_CLAUDE_USER_CONFIG_DIR"
      env -u QUIRKY_CLAUDE_INJECTED_CONFIG_DIR CLAUDE_CONFIG_DIR="$QUIRKY_CLAUDE_USER_CONFIG_DIR" claude $argv
      return $status
    end
    if test -z "$profile"
      env -u CLAUDE_CONFIG_DIR -u QUIRKY_CLAUDE_INJECTED_CONFIG_DIR claude $argv
      return $status
    end
    if not string match -qr '^(/|[A-Za-z]:)' -- "$profile"
      return 1
    end
    test -d "$profile"; or mkdir -p -m 700 -- "$profile" 2>/dev/null
    env CLAUDE_CONFIG_DIR="$profile" QUIRKY_CLAUDE_INJECTED_CONFIG_DIR="$profile" claude $argv
  end
end
set -e __quirky_claude_type
